# Audio-i Sound Pass Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** FTLite plays real sound: Kenney effects for combat, doors, consoles, fire, the engines and the jump; two Scott Buckley tracks that crossfade with the fight; and Master, Music and Effects volumes. The silent Single player duel gets sound too.

**Architecture:**
- The effects and tracks are committed under `public/assets/raw/` and listed in the manifest with `sfx.*` and `music.*` names.
- `loadAssets` hands their URLs to `app/audio.ts`, which fetches and decodes them in the background with plain Web Audio. It plays sample "requests" through per-voice panners into an effects bus, and runs a pure, time-injected music state machine on a music bus.
- Pure models in the new `src/renderer/src/audio/` folder turn existing duel `ViewEvent`s, tactical and interior state diffs, and sector cues into those requests and a per-frame status report. No sim, net or codec file changes.

**Tech Stack:** TypeScript 5.8, Web Audio (no library), Pixi.js 8.19 ticker, React 19 + zustand 5 (Settings), zod (manifest), vitest 3.2, Electron 37, playwright-core (scratchpad only, for the decode check).

**Spec:** `docs/superpowers/specs/2026-11-17-audio-i-design.md`

```ts
// Audio-i spec sections 1-3: the sound bank. Which Kenney file backs each effect key, the manifest names they bind, the
// music names, and the request shape the views hand the engine. Pure data: no Web Audio, no sim.

export type Pack = 'sci' | 'ui';
export const PACK_DIRS: Readonly<Record<Pack, string>> = { sci: 'raw/kenney-sci-fi-sounds', ui: 'raw/kenney-interface-sounds' };
interface SfxDef { readonly pack: Pack; readonly stem: string; readonly numbers: readonly number[] }

/** Each effect key: its pack, the Kenney file stem, and the file numbers of its variants (variant n is numbers[n]). */
export const SFX = {
  laserSmall: { pack: 'sci', stem: 'laserSmall', numbers: [0, 1, 2, 3, 4] },
  laserLarge: { pack: 'sci', stem: 'laserLarge', numbers: [0, 1, 2, 3, 4] },
  impactMetal: { pack: 'sci', stem: 'impactMetal', numbers: [0, 1, 2, 3, 4] },
  explosionCrunch: { pack: 'sci', stem: 'explosionCrunch', numbers: [0, 1, 2, 3, 4] },
  lowBoom: { pack: 'sci', stem: 'lowFrequency_explosion', numbers: [0, 1] },
  forceField: { pack: 'sci', stem: 'forceField', numbers: [0, 1, 2] },
  doorOpen: { pack: 'sci', stem: 'doorOpen', numbers: [0, 1, 2] },
  doorClose: { pack: 'sci', stem: 'doorClose', numbers: [0, 1, 2] },
  thruster: { pack: 'sci', stem: 'thrusterFire', numbers: [0, 1, 2] },
  engineHum: { pack: 'sci', stem: 'spaceEngineLow', numbers: [0] },
  jumpCharge: { pack: 'sci', stem: 'spaceEngineSmall', numbers: [0] },
  alarm: { pack: 'ui', stem: 'error', numbers: [4, 5] },
  ready: { pack: 'ui', stem: 'confirmation', numbers: [1, 2, 3] },
  upgrade: { pack: 'ui', stem: 'maximize', numbers: [6, 7, 8] },
  callout: { pack: 'ui', stem: 'question', numbers: [1, 2, 3] },
  console: { pack: 'ui', stem: 'select', numbers: [1, 2, 3] }
} as const satisfies Readonly<Record<string, SfxDef>>;
export type SfxKey = keyof typeof SFX;
export const SFX_KEYS = Object.keys(SFX) as SfxKey[];

export interface SfxFile { name: string; file: string; key: SfxKey; index: number }
export function sfxName(key: SfxKey, index: number): string { return `sfx.${key}.${index}`; }
/** Every effect file, in table order: its manifest name and its path under public/assets. */
export function sfxFiles(): SfxFile[] {
  return SFX_KEYS.flatMap((key) => {
    const d: SfxDef = SFX[key];
    return d.numbers.map((n, index) => ({ name: sfxName(key, index), file: `${PACK_DIRS[d.pack]}/${d.stem}_${String(n).padStart(3, '0')}.ogg`, key, index }));
  });
}

/** The two tracks (spec section 5): Cobalt is calm, Time and Space is battle. */
export const MUSIC_CALM = 'music.calm', MUSIC_BATTLE = 'music.battle';
export const MUSIC_FILES: Readonly<Record<string, string>> = { [MUSIC_CALM]: 'raw/scott-buckley/cobalt.mp3', [MUSIC_BATTLE]: 'raw/scott-buckley/time-and-space.mp3' };
export const AUDIO_ASSET_NAMES: readonly string[] = [...sfxFiles().map((f) => f.name), MUSIC_CALM, MUSIC_BATTLE];

export function isAudioName(name: string | undefined): boolean { return name !== undefined && (name.startsWith('sfx.') || name.startsWith('music.')); }
/** `sfx.<key>.<n>` back to its key and variant; null for anything else. */
export function sfxKeyOf(name: string): { key: SfxKey; index: number } | null {
  const m = /^sfx\.([A-Za-z]+)\.(\d+)$/.exec(name);
  if (!m || !Object.prototype.hasOwnProperty.call(SFX, m[1]!)) return null;
  const key = m[1] as SfxKey, index = Number(m[2]);
  return index < SFX[key].numbers.length ? { key, index } : null;
}

/** One sample to play (spec section 3). Times are seconds from now; `dur` cuts the sample with a short fade; `rate` is the
 *  playback rate (1 = as recorded), ramped to `rateTo` over `sweep` (default: the voice's length), and so is the filter's
 *  frequency from `from` to `to`; `variant` pins a variant instead of rotating; `detune` (default true) adds the random
 *  pitch variation. */
export interface SoundRequest {
  key: SfxKey;
  at?: number; gain?: number; pan?: number; dur?: number;
  rate?: number; rateTo?: number; sweep?: number;
  filter?: { type: 'lowpass' | 'bandpass'; from: number; to?: number; q?: number };
  variant?: number; detune?: boolean;
}
```

```ts
// Audio-i spec section 3: the pure parts of playing a sample. The variant pick, the detune, the pan and the per-key voice
// cap. Every random draw comes from the caller (the app's random source), never from the sim RNG.

export const VOICE_CAP = 4, PITCH_SPREAD_CENTS = 50, PAN_WIDTH = 0.8, PAN_WORLD_UNITS = 600;

const clamp = (v: number, lo: number, hi: number): number => (Number.isFinite(v) ? Math.min(hi, Math.max(lo, v)) : 0);

/** A variant of `key` out of `count`: random, but never the key's last pick when there is a choice. Remembers the pick. */
export function pickVariant(last: Map<string, number>, key: string, count: number, random: () => number): number {
  if (count <= 1) { last.set(key, 0); return 0; }
  const prev = last.get(key);
  let v: number;
  if (prev === undefined || prev >= count) v = Math.min(count - 1, Math.floor(random() * count));
  else { const r = Math.min(count - 2, Math.floor(random() * (count - 1))); v = r >= prev ? r + 1 : r; }
  last.set(key, v);
  return v;
}

/** A random detune in cents, within ±spread. */
export function detuneCents(random: () => number, spread = PITCH_SPREAD_CENTS): number { return (random() * 2 - 1) * spread; }

/** Pan from a screen x across a viewport `width` px wide: −PAN_WIDTH at the left edge, +PAN_WIDTH at the right. */
export function screenPan(sx: number, width: number): number { return width > 0 ? clamp((sx / width) * 2 - 1, -1, 1) * PAN_WIDTH : 0; }
/** Pan from a world-x offset from the listener (the crew ship), for a view with no map camera. */
export function worldPan(dx: number): number { return clamp(dx / PAN_WORLD_UNITS, -1, 1) * PAN_WIDTH; }
/** Pan from a tile column across a ship `width` tiles wide (interior doors and consoles). */
export function tilePan(column: number, width: number): number { return width > 0 ? screenPan(column + 0.5, width) : 0; }

export interface Voice { id: number; key: string; ends: number }
export interface VoicePool { next: number; voices: Voice[] }
export function createVoicePool(): VoicePool { return { next: 1, voices: [] }; }

/** Admit a voice of `key` that starts at `now` and ends at `ends` (seconds). Ended voices are forgotten; when `key` already
 *  has `cap` live voices, the oldest are returned in `steal` for the caller to stop, so the newest always sounds. */
export function admitVoice(pool: VoicePool, key: string, now: number, ends: number, cap = VOICE_CAP): { id: number; steal: number[] } {
  pool.voices = pool.voices.filter((v) => v.ends > now);
  const mine = pool.voices.filter((v) => v.key === key);
  const steal = mine.slice(0, Math.max(0, mine.length - (cap - 1))).map((v) => v.id);
  if (steal.length) pool.voices = pool.voices.filter((v) => !steal.includes(v.id));
  const id = pool.next++;
  pool.voices.push({ id, key, ends });
  return { id, steal };
}
```

```ts
/** One tone: oscillator type, start/end frequency (a sweep when they differ), offset and duration in seconds, peak gain. */

/** The three volumes, each 0 (off) to 1 (Audio-i spec section 6). */
export interface Volumes { master: number; music: number; effects: number }
/** What the views hand the audio (the duel scene, the sector session). */
export interface AudioOut {
  /** A cue: its sample recipe, or its synth tones when the recipe is null or not all decoded. */
  play(cue: Cue): void;
  /** Sample requests from a view's pure sound model. */
  sounds(list: readonly SoundRequest[]): void;

export interface Audio extends AudioOut {
  unlock(): void;
  suspend(): void;
  setVolumes(v: Volumes): void;
  /** Fetch and decode the audio entries in the background (spec section 2): the effects first, then the music. */
  load(sources: readonly SoundSource[], fetchBytes: (url: string) => Promise<ArrayBuffer>): Promise<void>;
}
export interface AudioOptions { context?: () => AudioContext | null; random?: () => number; log?: (m: string) => void }

function defaultContext(): AudioContext | null {
  const Ctor = (globalThis as { AudioContext?: typeof AudioContext }).AudioContext;
  return Ctor ? new Ctor() : null;
}
const level = (v: number): number => Math.max(0, Math.min(1, Number.isFinite(v) ? v : 1));
const panOf = (v: number | undefined): number => (v !== undefined && Number.isFinite(v) ? Math.max(-1, Math.min(1, v)) : 0);

export function createAudio(options: AudioOptions = {}): Audio {
  const random = options.random ?? Math.random, log = options.log ?? ((m: string): void => console.info(m));
  let ctx: AudioContext | null = null;
  try { ctx = (options.context ?? defaultContext)(); } catch { ctx = null; }
  if (!ctx) return { play() {}, sounds() {}, unlock() {}, suspend() {}, setVolumes() {}, async load() {} };
  const c = ctx;
  const safe = (f: () => void): void => { try { f(); } catch { /* audio unavailable: never disturb the game */ } };

  // The buses (spec section 3): voice -> panner -> effects -> master -> destination; music -> music bus -> master.
  const master = c.createGain(), musicBus = c.createGain(), effects = c.createGain(), synthBus = c.createGain();
  master.connect(c.destination); musicBus.connect(master); effects.connect(master); synthBus.connect(effects);
  master.gain.value = 1; musicBus.gain.value = MUSIC_LEVEL; effects.gain.value = EFFECTS_LEVEL; synthBus.gain.value = SYNTH_GAIN;

  const buffers = new Map<SfxKey, AudioBuffer[]>(), tracks = new Map<string, AudioBuffer>();
  const pool = createVoicePool(), live = new Map<number, AudioBufferSourceNode>(), lastVariant = new Map<string, number>();

  function synth(cue: Cue, t0: number): void {
    for (const t of CUES[cue]) {
      const osc = c.createOscillator(), g = c.createGain();
      osc.type = t.type; osc.frequency.setValueAtTime(t.f0, t0 + t.at);
      if (t.f1 !== t.f0) osc.frequency.linearRampToValueAtTime(t.f1, t0 + t.at + t.dur);
      g.gain.setValueAtTime(t.gain, t0 + t.at); g.gain.exponentialRampToValueAtTime(0.001, t0 + t.at + t.dur);   // fast decay, no click at the end
      osc.connect(g); g.connect(synthBus); osc.start(t0 + t.at); osc.stop(t0 + t.at + t.dur + 0.02);
    }
  }

  /** One sample voice (spec section 3): variant rotation, detune, filter sweep, cut, pan, the per-key cap. */
  function playRequest(r: SoundRequest, t0: number): void {
    const list = buffers.get(r.key);
    if (!list || list.length === 0) return;
    const v = r.variant !== undefined ? Math.max(0, Math.min(r.variant, list.length - 1)) : pickVariant(lastVariant, r.key, list.length, random);
    const buf = list[v]!, rate = r.rate ?? 1, start = t0 + (r.at ?? 0);
    const len = Math.min(r.dur ?? Number.POSITIVE_INFINITY, buf.duration / rate), sweep = r.sweep ?? len;
    const { id, steal } = admitVoice(pool, r.key, start, start + len);
    for (const s of steal) { try { live.get(s)?.stop(); } catch { /* already stopped */ } live.delete(s); }
    const src = c.createBufferSource(); src.buffer = buf;
    src.playbackRate.setValueAtTime(rate, start);
    if (r.rateTo !== undefined) src.playbackRate.linearRampToValueAtTime(r.rateTo, start + sweep);
    if (r.detune !== false) src.detune.value = detuneCents(random);
    let tail: AudioNode = src;
    if (r.filter) {
      const f = c.createBiquadFilter(); f.type = r.filter.type; f.frequency.setValueAtTime(r.filter.from, start);
      if (r.filter.to !== undefined) f.frequency.exponentialRampToValueAtTime(r.filter.to, start + sweep);
      if (r.filter.q !== undefined) f.Q.value = r.filter.q;
      tail.connect(f); tail = f;
    }
    const g = c.createGain(), peak = r.gain ?? 1;
    g.gain.setValueAtTime(peak, start);
    if (r.dur !== undefined) { g.gain.setValueAtTime(peak, Math.max(start, start + len - CUT_FADE_S)); g.gain.linearRampToValueAtTime(0, start + len); }
    const p = c.createStereoPanner(); p.pan.value = panOf(r.pan);
    tail.connect(g); g.connect(p); p.connect(effects);
    src.onended = () => { live.delete(id); };
    src.start(start); src.stop(start + len + 0.02);
    live.set(id, src);
  }

  return {
    play(cue) {
      safe(() => {
        const recipe = CUE_RECIPES[cue], t0 = c.currentTime;
        if (recipe && recipe.every((r) => (buffers.get(r.key)?.length ?? 0) > 0)) for (const r of recipe) playRequest(r, t0);
        else synth(cue, t0);
      });
    },
    sounds(list) { safe(() => { const t0 = c.currentTime; for (const r of list) playRequest(r, t0); }); },
    unlock() { safe(() => { void c.resume().catch(() => { /* rejected resume: ignore */ }); }); },
    suspend() { safe(() => { void c.suspend().catch(() => { /* rejected suspend: ignore */ }); }); },
    setVolumes(v) {
      safe(() => { master.gain.value = level(v.master); musicBus.gain.value = MUSIC_LEVEL * level(v.music); effects.gain.value = EFFECTS_LEVEL * level(v.effects); });
    },
    async load(sources, fetchBytes) {
      const decode = async (s: SoundSource): Promise<AudioBuffer | null> => {
        try { return await c.decodeAudioData(await fetchBytes(s.url)); }
        catch (err) { console.warn(`[audio] could not load ${s.url}`, err); return null; }
      };
      let ok = 0;
      const effectList = sources.filter((s) => !s.name.startsWith('music.')), trackList = sources.filter((s) => s.name.startsWith('music.'));
      const decoded = await Promise.all(effectList.map(decode));
      const staged = new Map<SfxKey, AudioBuffer[]>();
      effectList.forEach((s, k) => {
        const buf = decoded[k], id = sfxKeyOf(s.name);
        if (!buf || !id) return;
        ok++;
        const list = staged.get(id.key) ?? []; list[id.index] = buf; staged.set(id.key, list);
      });
      for (const [key, list] of staged) buffers.set(key, list.filter((b): b is AudioBuffer => !!b));   // a failed middle variant leaves no hole
      await Promise.all(trackList.map(async (s) => { const buf = await decode(s); if (buf) { ok++; tracks.set(s.name, buf); } }));
      log(`[audio] decoded ${ok}/${sources.length} files`);
    }
  };
}
```

```ts
// Audio-i spec section 4 (c, d): the two loops' levels and the fire crackle's samples. Pure: no Web Audio; the random
// source is the caller's (the app's), never the sim RNG.

const clamp01 = (v: number): number => (Number.isFinite(v) ? Math.min(1, Math.max(0, v)) : 0);

/** The engine hum's gain and playback rate, from idle to full (tune by ear). */
export const HUM_IDLE_GAIN = 0.1, HUM_MAX_GAIN = 0.35, HUM_IDLE_RATE = 0.85, HUM_MAX_RATE = 1.35;
/** The crew ship's speed (world units a second) that counts as full; the duel's top speed is THRUST / DRAG = 800. */
export const HUM_FULL_SPEED = 400;
export const HUM_THRUST_SHARE = 0.6, HUM_SPEED_SHARE = 0.4, HUM_SMOOTH_S = 0.25;

/** What drives the hum: thrust held (any sign, 0..1), the crew ship's speed, and the sector's jump swell (0..1). */
export interface HumInput { thrust: number; speed: number; charge: number }
export function humLevel(i: HumInput): number {
  const move = HUM_THRUST_SHARE * clamp01(Math.abs(i.thrust)) + HUM_SPEED_SHARE * clamp01(i.speed / HUM_FULL_SPEED);
  return clamp01(Math.max(move, clamp01(i.charge)));
}
export function humTarget(i: HumInput): { gain: number; rate: number } {
  const l = humLevel(i);
  return { gain: HUM_IDLE_GAIN + (HUM_MAX_GAIN - HUM_IDLE_GAIN) * l, rate: HUM_IDLE_RATE + (HUM_MAX_RATE - HUM_IDLE_RATE) * l };
}

/** The fire crackle's gain: 0 with no fire, then per burning room to a cap; smoothed and band-passed in the engine. */
export const FIRE_GAIN_ONE = 0.15, FIRE_GAIN_STEP = 0.08, FIRE_GAIN_MAX = 0.4, FIRE_SMOOTH_S = 0.6, FIRE_FILTER_HZ = 2200, FIRE_FILTER_Q = 0.6;
export function fireGain(rooms: number): number {
  return rooms >= 1 ? Math.min(FIRE_GAIN_MAX, FIRE_GAIN_ONE + FIRE_GAIN_STEP * (Math.floor(rooms) - 1)) : 0;
}

/** The crackle (spec section 4c decision: synthesised, not a processed sample): sparse random pops over a quiet noise bed,
 *  both ends faded so the loop never clicks. */
export const CRACKLE_SECONDS = 2, CRACKLE_POPS_PER_S = 40, CRACKLE_POP_S = 0.004, CRACKLE_BED = 0.08, CRACKLE_EDGE_S = 0.01;
export function crackleSamples(rate: number, random: () => number, seconds = CRACKLE_SECONDS): Float32Array {
  const n = Math.round(rate * seconds), out = new Float32Array(n), p = CRACKLE_POPS_PER_S / rate, decay = Math.exp(-1 / (CRACKLE_POP_S * rate));
  let env = 0;
  for (let k = 0; k < n; k++) {
    if (random() < p) env = 0.4 + 0.5 * random();
    out[k] = (random() * 2 - 1) * (CRACKLE_BED + env);
    env *= decay;
  }
  const edge = Math.min(n >> 1, Math.round(rate * CRACKLE_EDGE_S));
  for (let k = 0; k < edge; k++) { const f = k / edge; out[k] = out[k]! * f; out[n - 1 - k] = out[n - 1 - k]! * f; }
  return out;
}
```

```ts
// Audio-i spec section 5: the music state machine. Pure and time-injected: the engine runs it once per frame with the
// latest report. Calm plays by default; engaged moves to battle at once with a 2 s equal-power crossfade; 8 s with no
// engagement goes back to calm; a pause fades the mix out (and holds the calm-down count); a big hit ducks it.
import type { HumInput } from './loops';

export const CROSSFADE_S = 2, CALM_AFTER_S = 8, PAUSE_FADE_S = 0.4, DUCK_LEVEL = 0.5, DUCK_S = 0.6;
/** A report older than this (seconds) means no session is drawing (the menu): idle. */
export const REPORT_STALE_S = 0.5;
/** The longest step one frame may take (seconds), so a hitch never jumps the fade. */
export const MAX_STEP_S = 0.25;
/** The loop points (seconds), constants for the owner to tune by ear; none were measured. Battle starts at the orchestral
 *  turn (about 2:05) each time it begins, and loops from there to BATTLE_LOOP_END_S (null: the track's end). */
export const CALM_LOOP_START_S = 0, BATTLE_START_S = 125, BATTLE_LOOP_START_S = 125;
export const BATTLE_LOOP_END_S: number | null = null;

/** One frame's report from the drawing session (the duel scene or the sector session). */
export interface AudioStatus { engaged: boolean; paused: boolean; duck: boolean; hum: HumInput | null; fires: number }
export const IDLE_STATUS: AudioStatus = { engaged: false, paused: false, duck: false, hum: null, fires: 0 };
export function liveStatus(report: { status: AudioStatus; at: number } | null, now: number): AudioStatus {
  return report && now - report.at <= REPORT_STALE_S ? report.status : IDLE_STATUS;
}

export interface MusicState { mode: 'calm' | 'battle'; x: number; quietFor: number; pause: number; duckLeft: number; last: number | null }
export function createMusic(): MusicState { return { mode: 'calm', x: 0, quietFor: 0, pause: 1, duckLeft: 0, last: null }; }
export interface MusicInput { now: number; engaged: boolean; paused: boolean; duck: boolean }
/** The gains to aim for: calm and battle (equal power), the mix (pause x duck), and whether a battle source should exist. */
export interface MusicMix { calm: number; battle: number; mix: number; battleOn: boolean }

export function stepMusic(s: MusicState, i: MusicInput): MusicMix {
  const dt = s.last === null ? 0 : Math.min(MAX_STEP_S, Math.max(0, i.now - s.last));
  s.last = i.now;
  if (i.engaged) { s.mode = 'battle'; s.quietFor = 0; }
  else if (s.mode === 'battle' && !i.paused) { s.quietFor += dt; if (s.quietFor >= CALM_AFTER_S) s.mode = 'calm'; }
  const step = dt / CROSSFADE_S;
  s.x = s.mode === 'battle' ? Math.min(1, s.x + step) : Math.max(0, s.x - step);
  const fade = dt / PAUSE_FADE_S;
  s.pause = i.paused ? Math.max(0, s.pause - fade) : Math.min(1, s.pause + fade);
  s.duckLeft = i.duck ? DUCK_S : Math.max(0, s.duckLeft - dt);
  const calm = s.x === 0 ? 1 : Math.cos((s.x * Math.PI) / 2), battle = s.x === 1 ? 1 : Math.sin((s.x * Math.PI) / 2);
  return { calm, battle, mix: s.pause * (s.duckLeft > 0 ? DUCK_LEVEL : 1), battleOn: s.mode === 'battle' || s.x > 0 };
}
```

```ts
// The player's settings (main menu spec, 2026-11-16; Audio-i spec section 6): the Master, Music and Effects volumes. The
// same pattern as progressStore: read field by field, anything malformed falls back to the default, and storage failures
// never reach the game. A version 1 record (one volume) migrates: its volume becomes Master.
import type { RecordStorage } from './telemetryStore';
import type { Volumes } from './audio';

export const SETTINGS_KEY = 'ftlite-settings';
/** The volume choices Settings offers: Off, then quarters up to full. */
export const VOLUME_STEPS: readonly number[] = [0, 0.25, 0.5, 0.75, 1];
export const VOLUME_CHANNELS = ['master', 'music', 'effects'] as const;
export type VolumeChannel = (typeof VOLUME_CHANNELS)[number];
export const VOLUME_LABELS: Readonly<Record<VolumeChannel, string>> = { master: 'Master volume', music: 'Music', effects: 'Effects' };

export interface Settings extends Volumes { version: 2 }
export const DEFAULT_SETTINGS: Settings = { version: 2, master: 1, music: 1, effects: 1 };

const step = (v: unknown, fallback: number): number => (typeof v === 'number' && VOLUME_STEPS.includes(v) ? v : fallback);

export function loadSettings(storage: RecordStorage): Settings {
  try {
    const raw = JSON.parse(storage.getItem(SETTINGS_KEY) ?? 'null') as unknown;
    if (!raw || typeof raw !== 'object' || Array.isArray(raw)) return { ...DEFAULT_SETTINGS };
    const r = raw as Record<string, unknown>;
    if (r.version === 1) return { ...DEFAULT_SETTINGS, master: step(r.volume, DEFAULT_SETTINGS.master) };
    if (r.version !== 2) return { ...DEFAULT_SETTINGS };
    return { version: 2, master: step(r.master, DEFAULT_SETTINGS.master), music: step(r.music, DEFAULT_SETTINGS.music), effects: step(r.effects, DEFAULT_SETTINGS.effects) };
  } catch { return { ...DEFAULT_SETTINGS }; }
}

export function saveSettings(storage: RecordStorage, s: Settings): void {
  try { storage.setItem(SETTINGS_KEY, JSON.stringify(s)); } catch { /* storage unavailable or full */ }
}
```

```ts
// Audio-i spec section 4a: the duel's sounds, from the same per-frame ViewEvents the effects use (beside pushEvents, not
// inside it) and from two tactical state diffs (a missile launch, the round ending). Pure: the caller pans and plays.
import { hitsCrew, type ViewEvent } from '../render/tactical/fxModel';
import { HIT_FRESH_TICKS } from '../render/impactModel';
import type { Cue } from '../events/audioCues';
import type { SoundRequest } from './soundBank';

/** Copies of sim constants (types-only imports from sim/), pinned by tests/audioDuel.test.ts: the crew ship's slot and
 *  the outcomes Running and CrewWins. */
export const CREW = 0, OUTCOME_RUNNING = 0, OUTCOME_CREW_WINS = 1;
/** A hit lands this long after its shot (seconds). */
export const IMPACT_DELAY_S = 0.04;
/** World position to stereo pan (the scene's camera, or the crew-relative fallback on the interior view). */
export type Pan = (x: number, y: number) => number;
export interface EventSounds { sounds: SoundRequest[]; duck: boolean; combat: boolean }

export function duelEventSounds(events: readonly ViewEvent[], pan: Pan): EventSounds {
  const sounds: SoundRequest[] = [];
  let duck = false, combat = false;
  for (const e of events) {
    if (e.kind === 'laser') {
      combat = true;
      sounds.push({ key: e.ship === CREW ? 'laserSmall' : 'laserLarge', gain: 0.7, pan: pan(e.x1, e.y1) });
      const endPan = pan(e.x2, e.y2);
      if (e.stripped) sounds.push({ key: 'forceField', gain: 0.6, pan: endPan, at: IMPACT_DELAY_S, dur: 0.6 });
      else if (e.hit) sounds.push({ key: 'impactMetal', gain: 0.8, pan: endPan, at: IMPACT_DELAY_S });
    } else if (e.kind === 'missileEnd') {
      if (e.cause === 'edge' || e.cause === 'expired') continue;
      combat = true;
      sounds.push({ key: 'explosionCrunch', gain: e.cause === 'ship' ? 1 : 0.6, pan: pan(e.x, e.y) });
    } else {
      sounds.push({ key: 'impactMetal', gain: e.ship === CREW ? 0.6 : 0.3, rate: 0.8, pan: pan(e.x, e.y) });
    }
    if (hitsCrew(e)) { duck = true; sounds.push({ key: 'lowBoom', gain: 0.8, pan: 0 }); }
  }
  return { sounds, duck, combat };
}

/** What the state diff reads each frame: the tactical tick, the round, the outcome and each missile slot. */
export interface DuelWatchView { tick: number; round: number; outcome: number; missiles: readonly { active: boolean; x: number; y: number }[] }
export interface DuelWatch { tick: number; round: number; outcome: number; missiles: boolean[] }

/** A missile launch (a slot going active) and the round ending (victory or defeat). A missing baseline, a new round, a tick
 *  going backwards or a jump of more than HIT_FRESH_TICKS re-baselines silently (no replay after a resync or a rejoin). */
export function duelStateSounds(w: DuelWatch | null, v: DuelWatchView, pan: Pan): { watch: DuelWatch; sounds: SoundRequest[]; cues: Cue[] } {
  const watch: DuelWatch = { tick: v.tick, round: v.round, outcome: v.outcome, missiles: v.missiles.map((m) => m.active) };
  const sounds: SoundRequest[] = [], cues: Cue[] = [];
  if (!w || w.round !== v.round || v.tick < w.tick || v.tick - w.tick > HIT_FRESH_TICKS || w.missiles.length !== v.missiles.length) return { watch, sounds, cues };
  v.missiles.forEach((m, k) => { if (m.active && !w.missiles[k]) sounds.push({ key: 'thruster', gain: 0.5, dur: 0.7, pan: pan(m.x, m.y) }); });
  if (w.outcome === OUTCOME_RUNNING && v.outcome !== OUTCOME_RUNNING) cues.push(v.outcome === OUTCOME_CREW_WINS ? 'victory' : 'defeat');
  return { watch, sounds, cues };
}
```

```ts
// Audio-i spec section 4c: the ship interior's sounds from render-side state diffs, for the sector state and the duel's
// interior alike: a door opening or closing, sitting at a console, a room catching fire, and the burning-room count that
// drives the crackle loop. The same stale-replay guard as render/impactModel.ts.
import type { SimState } from '../sim/state';
import { HIT_FRESH_TICKS } from '../render/impactModel';
import { tilePan } from './voices';
import type { SoundRequest } from './soundBank';

/** The door states that pass crew (Open, HeldOpen): a copy of sim/grid's, pinned by tests/audioInterior.test.ts. */
export const OPEN_DOOR_STATES: readonly number[] = [1, 3];

/** What the diff reads: the tick, the per-edge door states, the per-room fire, the per-station occupant (-1 empty), and a
 *  pan per door edge and per station. */
export interface InteriorAudioView {
  tick: number; doors: ArrayLike<number>; fire: ArrayLike<number>; occupant: ArrayLike<number>;
  doorPan: readonly number[]; stationPan: readonly number[];
}
export interface InteriorWatch { tick: number; open: boolean[]; burning: boolean[]; seated: boolean[] }

/** A sim state as the diff reads it; `tick` defaults to the state's own (the duel passes its tactical tick). */
export function interiorAudioView(state: SimState, tick = state.tick): InteriorAudioView {
  const d = state.derived, w = d.width;
  return {
    tick, doors: state.doors.state, fire: state.rooms.fire, occupant: state.stations.occupant,
    doorPan: Array.from(d.edgeTile, (t) => tilePan(t % w, w)), stationPan: Array.from(d.stationTile, (t) => tilePan(t % w, w))
  };
}

export function interiorSounds(w: InteriorWatch | null, v: InteriorAudioView): { watch: InteriorWatch; sounds: SoundRequest[]; fires: number } {
  const open = Array.from(v.doors, (s) => OPEN_DOOR_STATES.includes(s)), burning = Array.from(v.fire, (f) => f > 0), seated = Array.from(v.occupant, (o) => o >= 0);
  const fires = burning.filter(Boolean).length, watch: InteriorWatch = { tick: v.tick, open, burning, seated };
  const fresh = !!w && v.tick >= w.tick && v.tick - w.tick <= HIT_FRESH_TICKS
    && w.open.length === open.length && w.burning.length === burning.length && w.seated.length === seated.length;
  if (!w || !fresh) return { watch, sounds: [], fires };
  const sounds: SoundRequest[] = [];
  open.forEach((o, e) => { if (o !== w.open[e]) sounds.push({ key: o ? 'doorOpen' : 'doorClose', gain: 0.6, pan: v.doorPan[e] ?? 0 }); });
  seated.forEach((s, k) => { if (s && !w.seated[k]) sounds.push({ key: 'console', gain: 0.5, pan: v.stationPan[k] ?? 0 }); });
  burning.forEach((b, r) => { if (b && !w.burning[r]) sounds.push({ key: 'thruster', gain: 0.45, dur: 0.5, filter: { type: 'lowpass', from: 1200 } }); });
  return { watch, sounds, fires };
}
```

```ts
// Audio-i spec sections 4-5: the duel's audio adapter, fed once per frame by the scene (Single player and the online duel
// alike): the event and state sounds, the round-end cue, and the music and loop report.
import type { AudioOut } from '../audio';
import type { DuelState } from '../../sim/duel/state';
import type { Axes } from '../../sim/tactical/state';
import type { ViewEvent } from '../../render/tactical/fxModel';
import { CREW, OUTCOME_RUNNING, duelEventSounds, duelStateSounds, type DuelWatch, type Pan } from '../../audio/duelSounds';
import { interiorAudioView, interiorSounds, type InteriorWatch } from '../../audio/interiorSounds';

export interface DuelAudioFrame { duel: DuelState; events: readonly ViewEvent[]; axes: Axes; paused: boolean; pan: Pan }

export function createDuelAudio(out: AudioOut) {
  let duelWatch: DuelWatch | null = null, inside: InteriorWatch | null = null;
  return {
    frame(f: DuelAudioFrame): void {
      const t = f.duel.tactical, ev = duelEventSounds(f.events, f.pan);
      const st = duelStateSounds(duelWatch, { tick: t.tick, round: f.duel.round, outcome: t.outcome, missiles: t.missiles }, f.pan);
      duelWatch = st.watch;
      const ins = interiorSounds(inside, interiorAudioView(f.duel.interior, t.tick));
      inside = ins.watch;
      const list = [...ev.sounds, ...st.sounds, ...ins.sounds];
      if (list.length) out.sounds(list);
      for (const cue of st.cues) out.play(cue);
      const ship = t.ships[CREW]!;
      // Engaged: the round is running (spec section 5). The hum: thrust as the Helm holds it (a non-Helm online seat sees no
      // axes, so only the speed part), and the crew ship's speed.
      out.status({ engaged: t.outcome === OUTCOME_RUNNING, paused: f.paused, duck: ev.duck, fires: ins.fires,
        hum: { thrust: Math.abs(f.axes.thrust) > 0 ? 1 : 0, speed: Math.hypot(ship.vx, ship.vy), charge: 0 } });
    },
    /** A new round or a rebuilt scene: the next frame is a fresh baseline. */
    reset(): void { duelWatch = null; inside = null; }
  };
}
export type DuelAudio = ReturnType<typeof createDuelAudio>;
```

```ts
// Audio-i spec sections 4d and 5: the sector's music rule and the hum's jump swell. Pure, read from the drawn state and
// the frame's cues.
import type { Cue } from '../events/audioCues';
import type { SimState } from '../sim/state';

/** Cues that mean a fight is on this frame. */
export const COMBAT_CUES: readonly Cue[] = ['shot', 'hit', 'impact'];

/** Engaged (spec section 5): an enemy ship is present, a boarder is aboard, or a shot, hit or impact cue sounded this frame. */
export function sectorEngaged(state: Pick<SimState, 'enemy' | 'boarders'>, cues: ReadonlySet<Cue>): boolean {
  if (state.enemy.present === 1) return true;
  for (const a of state.boarders.active) if (a) return true;
  return COMBAT_CUES.some((c) => cues.has(c));
}

/** The hum's swell while the jump drive charges (0..1); 0 once it is ready (1), when the ready chime takes over. */
export function chargeSwell(jumpCharge: number): number {
  return Number.isFinite(jumpCharge) && jumpCharge < 1 ? Math.max(0, jumpCharge) : 0;
}
```

```ts
// Audio-i spec sections 4c-4d and 5: the sector's audio adapter, fed once per drawn frame by the online sector session:
// the interior sounds (doors, consoles, fire) and the music and loop report. The cues themselves still play through
// playCue (now as samples, the jump as its layered recipe).
import type { SimState } from '../sim/state';
import type { Cue } from '../events/audioCues';
import type { AudioOut } from './audio';
import { interiorAudioView, interiorSounds, type InteriorWatch } from '../audio/interiorSounds';
import { chargeSwell, sectorEngaged } from '../audio/sectorSounds';

export interface SectorAudio { frame(state: SimState, paused: boolean, cues: ReadonlySet<Cue>): void; reset(): void }

export function createSectorAudio(out: AudioOut): SectorAudio {
  let inside: InteriorWatch | null = null;
  return {
    frame(state, paused, cues) {
      const r = interiorSounds(inside, interiorAudioView(state));
      inside = r.watch;
      if (r.sounds.length) out.sounds(r.sounds);
      // The impact cue ducks the music itself (DUCK_CUES), so the report carries no duck. The sector has no thrust: the hum
      // idles and swells while the drive charges.
      out.status({ engaged: sectorEngaged(state, cues), paused, duck: false, fires: r.fires, hum: { thrust: 0, speed: 0, charge: chargeSwell(state.run.jumpCharge) } });
    },
    reset() { inside = null; }
  };
}
```
